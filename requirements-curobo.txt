# g1_curobo env freeze (py3.11, numpy2, torch-cu128, cuRobo V2). The whole repo runs in
# this one conda env. Some pieces are NOT plain `pip install` and must be built in order
# BEFORE installing this file -- full copy-paste recipe in README.md "Replicating the env":
#   conda create -n g1_curobo python=3.11 -y
#   # 1. CUDA torch for the RTX 5090 (sm_120):
#   pip install torch==2.9.1 --index-url https://download.pytorch.org/whl/cu128
#   # 2. cuRobo V2 from source (CUDA ext built for sm_120; pin below for the exact commit):
#   git clone https://github.com/NVlabs/curobo ~/repos/curobo
#   CUDA_HOME=/usr/local/cuda-12.8 PATH=/usr/local/cuda-12.8/bin:$PATH \
#     TORCH_CUDA_ARCH_LIST="12.0" pip install -e ~/repos/curobo[cu12] --no-build-isolation
#   # 3. CycloneDDS python bindings built against the system C lib:
#   CYCLONEDDS_HOME=/opt/cyclonedds pip install cyclonedds==0.10.2
#   # 4. Unitree SDK (editable, --no-deps so it can't downgrade numpy):
#   git clone https://github.com/unitreerobotics/unitree_sdk2_python ~/repos/unitree_sdk2_python
#   pip install -e ~/repos/unitree_sdk2_python --no-deps
#   # 5. everything else (incl. msgpack/msgpack-numpy -- the GraspGenX/SAM3 ZMQ wire deps):
#   pip install -r requirements-curobo.txt
#   # 6. this package:
#   pip install -e .
# Run everything via the `use_conda g1_curobo` lane wrapper (PYTHONNOUSERSITE=1 isolation);
